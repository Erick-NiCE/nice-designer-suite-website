## Examples / With Chips Expansion

Usage:
```tsx
({
  storyTitle,
  invalidFirstTag,
  restoreFocusOnLastTagDeleted,
  ...args
}) => {
  const initialTags = () => (args.tagsSource || []).map((tag, i) => i === 0 && invalidFirstTag ? {
    ...tag,
    errorState: true,
    errorMessage: 'Invalid value'
  } : {
    ...tag
  });
  const [localTags, setLocalTags] = useState<SelectedTag[]>(initialTags);
  useEffect(() => {
    setLocalTags(initialTags());
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [args.tagsSource, invalidFirstTag]);
  return <div>
      <header>
        <h1 className="sol-screenreader-only">{storyTitle}</h1>
      </header>
      <main>
        <SOLChipGroup {...args} tagsSource={localTags} onTagsChanged={e => {
        setLocalTags([...e.tags]);
        action('tags changed')(e);
      }} onDeleteAllClicked={() => {
        setLocalTags([]);
        action('cleared all')();
      }} onTagDeleted={action('tag deleted')} onLastTagDeleted={() => {
        action('last tag deleted')();
        if (restoreFocusOnLastTagDeleted) document.getElementById('chip-group-next-control')?.focus();
      }} onViewMoreClicked={action('view more clicked')} onViewLessClicked={action('view less clicked')} />
        {restoreFocusOnLastTagDeleted && <button id="chip-group-next-control" type="button">
            Next control
          </button>}
        <p style={{
        fontFamily: 'SOL Sans',
        fontSize: '.875rem'
      }}>(Resize the browser window to see the number of visible tags change.)</p>
      </main>
    </div>;
}
```

## Examples / With Popover

Usage:
```tsx
({
  storyTitle,
  invalidFirstTag,
  restoreFocusOnLastTagDeleted,
  ...args
}) => {
  const initialTags = () => (args.tagsSource || []).map((tag, i) => i === 0 && invalidFirstTag ? {
    ...tag,
    errorState: true,
    errorMessage: 'Invalid value'
  } : {
    ...tag
  });
  const [localTags, setLocalTags] = useState<SelectedTag[]>(initialTags);
  useEffect(() => {
    setLocalTags(initialTags());
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [args.tagsSource, invalidFirstTag]);
  return <div>
      <header>
        <h1 className="sol-screenreader-only">{storyTitle}</h1>
      </header>
      <main>
        <SOLChipGroup {...args} tagsSource={localTags} onTagsChanged={e => {
        setLocalTags([...e.tags]);
        action('tags changed')(e);
      }} onDeleteAllClicked={() => {
        setLocalTags([]);
        action('cleared all')();
```
