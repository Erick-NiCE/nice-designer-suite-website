# Wizard (real react markup, Lyra theme)

Stylesheets to load after `base.css`: `css/Button.css`, `css/Icon.css`, `css/Tooltip.css`, `css/Wizard.css`.
Component classes: `sol-button`, `sol-icon`, `sol-icon-host`, `sol-icon-pointer-events-none`, `sol-icon-template-wrapper`, `sol-screenreader-only`, `sol-wizard-connector`, `sol-wizard-connector--default`, `sol-wizard-connector--in-progress`, `sol-wizard-connector-wrapper`, `sol-wizard-container`.

This is the markup Storybook renders, class names included. Copy it as is. Do not restyle it and do not rename its classes: the classes are how the audit recognizes a real component.

## Examples / Wizard

Usage:
```tsx
args => {
  const wizardRef = useRef<WizardHandle>(null);
  const [currentStepIndex, setCurrentStepIndex] = useState(0);
  const [completed, setCompleted] = useState<boolean[]>(STEP_IDS.map((_, index) => index === 0));
  const handleStepChanged = (step: StepChangedData) => {
    setCurrentStepIndex(step.stepIndex);
  };
  const markCurrentStepComplete = () => {
    setCompleted(prev => prev.map((value, index) => index === currentStepIndex ? true : value));
  };
  const stepLabel = (stepNumber: number) => args.isLongLabel ? `long long long long Step ${stepNumber}` : `Step ${stepNumber}`;
  return <main>
      <h1 className="sol-screenreader-only">Wizard</h1>
      <div className="wrapper-item">
        <div style={{
        display: 'flex',
        justifyContent: 'flex-end',
        marginBottom: '10px',
        position: 'relative',
        right: '10px'
      }}>
          <SOLButton variant="primary" size="medium" icon={BackIcon} onButtonClick={() => wizardRef.current?.prev()}>
            Back
          </SOLButton>
          &nbsp;&nbsp;
          <SOLButton variant="primary" size="medium" iconEnd={NextIcon} onButtonClick={() => wizardRef.current?.next()}>
            Next
          </SOLButton>
        </div>
        <div style={{
        padding: '30px',
        width: args.containerWidth || undefined,
        boxShadow: '0 4px 8px 0 rgba(0, 0, 0, 0.2), 0 6px 20px 0 rgba(0, 0, 0, 0.19)'
      }}>
          <SOLWizard ref={wizardRef} id="wizard2" allowJumping={args.allowJumping} onStepChanged={handleStepChanged}>
            {STEP_IDS.map((stepId, index) => <SOLStep key={stepId} id={stepId} label={stepLabel(index + 1)} completed={completed[index]} disabled={!!args.isDisabled && index === 2} />)}
          </SOLWizard>
        </div>
      </div>

      <div className="step-content" style={{
      padding: '25px'
    }}>
        Step {currentStepIndex + 1} contents
        <br />
        <br />
        <SOLButton variant="primary" id="step_complete" onButtonClick={markCurrentStepComplete}>
          <span>Step Complete</span>
        </SOLButton>
      </div>
    </main>;
}
```
