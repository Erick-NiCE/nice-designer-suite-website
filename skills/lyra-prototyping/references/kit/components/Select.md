## Examples / Custom Template with Icons

Usage:
```tsx
args => {
  const [single, setSingle] = useState<OptionType | OptionType[]>('yellow');
  const [multi, setMulti] = useState<OptionType | OptionType[]>(['yellow', 'green', 'blue', 'selected white']);
  const storyOptions = args.useProjectedContent ? optionsWithProjection : optionsWithProjection.map(o => ({
    ...o,
    icon: './assets/application-icons.svg#icon-as-date'
  }));
  const projected = (scope: {
    ariaLabel: string;
  }) => <div style={{
    display: 'flex',
    alignItems: 'center',
    gap: '0.5rem'
  }}>
      <SOLIcon icon="./assets/application-icons.svg#icon-as-date" width="1rem" />
      <div style={{
      display: 'flex',
      flexDirection: 'column'
    }}>
        <span>{scope.ariaLabel}</span>
        <span style={{
        fontSize: '0.75rem',
        color: 'var(--sol-color-text-secondary)'
      }}>{args.secondaryRowText}</span>
      </div>
    </div>;
  return <main>
      <div style={{
      width: 300,
      marginBottom: '2rem'
    }}>
        <SOLSelect id="single-custom-template" options={storyOptions} selection={single} isMultiple={false} projectedContent={args.useProjectedContent ? projected : undefined} onSelectionChange={v => {
        setSingle(v);
        action('single selection changed')(v);
      }} />
      </div>
      <div style={{
      width: 300
    }}>
        <SOLSelect id="multi-custom-template" options={storyOptions} selection={multi} isMultiple={true} projectedContent={args.useProjectedContent ? projected : undefined} showFooter={true} onSelectionChange={v => {
        setMulti(v);
        action('multi selection changed')(v);
      }} />
      </div>
    </main>;
}
```
