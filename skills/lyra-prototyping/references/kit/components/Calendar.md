## Examples / Multi Single Selection

Usage:
```tsx
args => {
  const [value, setValue] = useState<CalendarValue>(() => initialSingle(args.data, args.type));
  const [viewKey, setViewKey] = useState(() => monthKey(value));
  // Re-seed when the `data` control changes, mirroring Angular's [(ngModel)]="data" writeValue.
  useEffect(() => {
    const seeded = initialSingle(args.data, args.type);
    setValue(seeded);
    setViewKey(monthKey(seeded));
  }, [args.type, args.data]);
  return <div>
      <header>
        <h1 id="calendar-story-heading" className="sol-screenreader-only">
          {args.heading}
        </h1>
      </header>
      <main>
        <div style={{
        width: '328px',
        height: '384px'
      }}>
          <SOLCalendar key={viewKey} type={args.type} daysDisabled={args.daysDisabled} min={coerceToDate(args.min)} max={coerceToDate(args.max)} startingDay={args.startingDay} specialDates={args.type === 'single-selection' ? args.specialDates : []} ariaLabel={args.ariaLabel} ariaLabelledby={args.ariaLabelledby} ariaDescribedby={args.ariaDescribedby} value={value} onChange={setValue} />
          {args.type === 'single-selection' && <div>{fmt(value as Date)}</div>}
          {args.type === 'multi-single-selection' && (value as Date[]).map((d, i) => <div key={i}>{fmt(d)}</div>)}
        </div>
      </main>
    </div>;
}
```

## Examples / Multi Week Selection

Usage:
```tsx
args => {
  const [value, setValue] = useState<CalendarValue>(() => initialWeek(args.data, args.type));
  const typeRef = useRef(args.type);
  useEffect(() => {
    if (typeRef.current !== args.type) {
      typeRef.current = args.type;
      setValue(initialWeek(args.data, args.type));
    }
  }, [args.type, args.data]);
  return <div>
      <header>
        <h1 id="calendar-story-heading" className="sol-screenreader-only">
          {args.heading}
        </h1>
      </header>
      <main>
        <div style={{
        width: '328px',
        height: '384px'
      }}>
          <SOLCalendar type={args.type} daysDisabled={args.daysDisabled} min={coerceToDate(args.min)} max={coerceToDate(args.max)} startingDay={args.startingDay} specialDates={args.specialDates} ariaLabel={args.ariaLabel} ariaLabelledby={args.ariaLabelledby} value={value} onChange={setValue} />
          {args.type === 'week-selection' && <div>
              {fmt((value as DateRange)?.start)} - {fmt((value as DateRange)?.end)}
            </div>}
          {args.type === 'multi-week-selection' && (value as DateRange[]).map((d, i) => <div key={i}>
                {fmt(d?.start)} - {fmt(d?.end)}
              </div>)}
        </div>
      </main>
    </div>;
}
```

```tsx
      <header>
```
