## Examples / Non Linear Range Slider

Usage:
```tsx
args => {
  const [dragValues, setDragValues] = useState<number[]>([1800, 7200]);
  const tooltipFormatter = useCallback((seconds: number): string => {
    if (seconds >= 3600) {
      const hours = Math.floor(seconds / 3600);
      const minutes = Math.floor(seconds % 3600 / 60);
      return hours.toString().padStart(2) + 'h ' + minutes.toString().padStart(2) + 'm ';
    } else if (seconds >= 60) {
      const minutes = Math.round(seconds / 60);
      return minutes.toString().padStart(2) + 'm ';
    } else {
      return Math.round(seconds) + 's';
    }
  }, []);
  const range: Range = {
    min: [0, 1],
    '14%': [30, 1],
    '27%': [60, 60],
    '30%': [300, 60],
    '40%': [900, 60],
    '50%': [1800, 60],
    '70%': [3600, 60],
    max: [7200]
  };
  const pips: Pips = {
    mode: 'steps',
    filter: function nonLinearFilterPips(value: number, type: PipsType): PipsType {
      if (type === 0) {
        return value > 60 ? 1 : 0;
      }
      return value === 0 || value === 30 || value === 60 || value === 300 || value === 900 || value === 1800 || value === 3600 || value === 7200 ? 1 : 0;
    },
    density: 60,
    format: {
      to(seconds: number): string {
        if (seconds >= 3600) {
          const hours = Math.floor(seconds / 3600);
          const minutes = Math.floor(seconds % 3600 / 60);
          return hours.toString().padStart(2) + 'h ' + minutes.toString().padStart(2) + 'm ';
        } else if (seconds >= 60) {
          const minutes = Math.round(seconds / 60);
          return minutes.toString().padStart(2) + 'm ';
        } else {
          return Math.round(seconds) + 's';
        }
      },
      from(formattedTime: string): number {
        const parts = formattedTime.split(' ');
        if (parts.length === 2) {
          const value1 = parseInt(parts[0], 10);
          const value2 = parseInt(parts[1], 10);
          if (parts[1] === 'h') {
            return value1 * 3600 + value2 * 60;
          } else if (parts[1] === 'm') {
            return value1 * 60 + value2;
          }
          return 0;
        } else {
          return parseInt(parts[0], 10);
        }
      }
    }
  };
  return <>
      <header>
        <h1 className="sol-screenreader-only">Non-Linear Range Slider</h1>
      </header>
      <main>
        <div>
          <SOLRangeSlider {...args} value={dragValues} range={range} pips={pips} tooltipFormatter={tooltipFormatter} onChange={values => setDragValues(values)} />
          <div className="common-text" style={{
          marginTop: '24px'
        }}>
            NgModel Values: <span>{dragValues.map(v => tooltipFormatter(v).trim()).join(',')}</span>
          </div>
        </div>
      </main>
    </>;
}
```

## Examples / Range Slider With Min Max

Usage:
```tsx
args => {
  const [values1, setValues1] = useState<number[]>([10, 60]);
  const tooltipFormatter = useCallback((value: number): string => {
    const roundedValue = Math.round(value);
    const minute = Math.floor(roundedValue / 60);
    const second = Math.round(roundedValue % 60);
    const minuteStr = minute <= 9 ? '0' + minute.toString() : minute.toString();
    const secondsStr = second <= 9 ? '0' + second.toString() : second.toString();
    return minuteStr + ':' + secondsStr;
  }, []);
  const markerLabelFormatter = useCallback((value: number, maxRange: number, minRange: number): string => {
    const roundedValue = Math.round(value);
    const minute = Math.floor(roundedValue / 60);
    const second = Math.round(roundedValue % 60);
    if ((minute % 5 === 0 || roundedValue === maxRange || roundedValue === minRange) && second === 0) {
      return minute.toString() + 'm';
    }
    return '';
  }, []);
  return <>
      <header>
        <h1 className="sol-screenreader-only">Range Slider With Min/Max</h1>
      </header>
      <main>
        <div>
          <SOLRangeSlider {...args} value={values1} tooltipFormatter={tooltipFormatter} markerLabelFormatter={markerLabelFormatter} onChange={values => {
          setValues1(values);
          action('Change Event')(values);
        }} onEnd={values => action('End Event')(values)} />
          <div className="common-text" style={{
          marginTop: '24px'
        }}>
            NgModel Values: <span>{values1.join(',')}</span>
          </div>
        </div>
      </main>
    </>;
}
```
